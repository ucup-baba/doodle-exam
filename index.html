/**
 * DOODLE EXAM - BACKEND GOOGLE APPS SCRIPT (Code.gs)
 * Ditautkan ke Google Spreadsheet: 13lFYNxpwdyA2M1OiyxhETYX1gnlyvF5lhaCZjWZaUZc
 * Materi: Teorema Pythagoras Kelas VIII SMP/MTs
 */
const SPREADSHEET_ID = '13lFYNxpwdyA2M1OiyxhETYX1gnlyvF5lhaCZjWZaUZc';
const SHEET_HASIL = 'Hasil_Ujian';
const SHEET_DETAIL = 'Detail_Jawaban';

function getSpreadsheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  } catch (err) {
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function doGet(e) {
  setupSheetsIfNeeded();
  return ContentService.createTextOutput("Backend Doodle Exam Google Sheets Aktif & Terhubung!")
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const result = simpanHasilUjianKeSheets(payload);
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function setupSheetsIfNeeded() {
  const ss = getSpreadsheet();
  let sheetHasil = ss.getSheetByName(SHEET_HASIL);
  if (!sheetHasil) {
    sheetHasil = ss.insertSheet(SHEET_HASIL);
    sheetHasil.appendRow([
      'Timestamp', 'Nama Siswa', 'Kelas', 'No Absen', 'Kode Ujian',
      'Durasi', 'Skor PG (30)', 'Skor Mencocokkan (12)', 'Skor Kompleks (18)',
      'Total Objektif (60)', 'Skor Esai (40)', 'Total Skor (100)', 'Kutipan Sesi'
    ]);
    sheetHasil.getRange(1, 1, 1, 13).setFontWeight('bold').setBackground('#fef08a');
  }

  let sheetDetail = ss.getSheetByName(SHEET_DETAIL);
  if (!sheetDetail) {
    sheetDetail = ss.insertSheet(SHEET_DETAIL);
    sheetDetail.appendRow(['Timestamp', 'Nama Siswa', 'Kelas', 'No Absen', 'JSON Jawaban Lengkap']);
    sheetDetail.getRange(1, 1, 1, 5).setFontWeight('bold').setBackground('#bae6fd');
  }
}

function simpanHasilUjianKeSheets(data) {
  setupSheetsIfNeeded();
  const ss = getSpreadsheet();
  const sheetHasil = ss.getSheetByName(SHEET_HASIL);
  const sheetDetail = ss.getSheetByName(SHEET_DETAIL);
  const now = new Date();

  const mins = Math.floor((data.durationSeconds || 0) / 60);
  const secs = (data.durationSeconds || 0) % 60;
  const durasiFormatted = mins + ' menit ' + secs + ' detik';

  sheetHasil.appendRow([
    now,
    data.student.name,
    data.student.className,
    data.student.absent,
    data.student.token,
    durasiFormatted,
    data.scores.mcqScore,
    data.scores.matchScore,
    data.scores.complexScore,
    data.scores.objectiveTotal,
    data.scores.essayScore,
    data.scores.grandTotal,
    data.quoteUsed
  ]);

  sheetDetail.appendRow([
    now,
    data.student.name,
    data.student.className,
    data.student.absent,
    JSON.stringify(data.answers)
  ]);

  return { status: 'success', timestamp: now.toISOString() };
}
